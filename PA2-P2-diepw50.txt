/*
   Model parameter
*/ 
#define N 11
#define M (N/2)
int position[N];

/*
   position[i] is 
      0: no frog in the position;
      1: green frogs in the position;
      2: yellow frogs in the position 

   green frog moves from left to right
   yellow frog moves from right to left
*/

/*
   Use the move function:
   pos1: current position of the frog
   pos2: new position of the frog after the move
   frog: either 1 or 2 depending on its color
*/ 
#define move(pos1, pos2, frog) atomic{ printf("Frog %d moves from %d to %d\n", frog, pos1, pos2); position[pos1] = 0; position[pos2] = frog; }

/*
   Write the model
   1. For whatever reason, none of the frogs want to turn back. So they can only move forward.
   2. Furthermore, they can either jump over one frog or can jump to the rock right in front of
   them.
   3. Each rock can hold just one frog.
   4. None of the frogs want to get wet.
*/

init {
   /* set up the array */
   int i = 0;
   i = 0;
   do
   :: i < M  -> position[i] = 1; i++
   :: i >= M -> break
   od;

   position[M] = 0;
   i = M + 1;
   
   do
   :: i < N  -> position[i] = 2; i++
   :: i >= N -> break
   od;

   run move_frogs()
}

proctype move_frogs() {
    int i;
    do
    /* green step */
    :: atomic {
         i = 0;
         do
         :: i < N-1 ->
            if
            :: position[i] == 1 && position[i+1] == 0 ->
                  move(i, i+1, 1); break
            :: else -> skip
            fi;
            i++
         :: i >= N-1 -> break
         od
      }

    /* green jump */
    :: atomic {
         i = 0;
         do
         :: i < N-2 ->
            if
            :: position[i] == 1 && position[i+1] == 2 && position[i+2] == 0 ->
                  move(i, i+2, 1); break
            :: else -> skip
            fi;
            i++
         :: i >= N-2 -> break
         od
      }

    /* yellow step */
    :: atomic {
      i = N-1;
         do
         :: i > 0 ->
            if
            :: position[i] == 2 && position[i-1] == 0 ->
                  move(i, i-1, 2); break
            :: else -> skip
            fi;
            i--
         :: i <= 0 -> break
         od
      }

   /* yellow jump */
   :: atomic {
      i = N-1;
         do
         :: i > 1 ->
            if
            :: position[i] == 2 && position[i-1] == 1 && position[i-2] == 0 ->
                  move(i, i-2, 2); break
            :: else -> skip
            fi;
            i--
         :: i <= 1 -> break
         od
      }
   od
}
